  <div id="tablero" aria-label="Tablero de ajedrez"></div>

  <div id="estado">Selecciona una pieza para moverla.</div>

  <button id="reiniciar">Reiniciar partida</button>

  <script>
    const tablero = document.getElementById("tablero");
    const estado = document.getElementById("estado");

 
    ];

    let piezas;
    let seleccion = null;

    const blancas = "♙♖♘♗♕♔";

    function dibujar() {
      tablero.innerHTML = "";

      for (let fila = 0; fila < 8; fila++) {
        for (let col = 0; col < 8; col++) {
          const casilla = document.createElement("button");
          const pieza = piezas[fila][col];

          casilla.type = "button";
          casilla.className =
            "casilla " +
            ((fila + col) % 2 === 0 ? "clara" : "oscura");

          casilla.setAttribute(
            "aria-label",
            String.fromCharCode(97 + col) + (8 - fila) +
            (pieza ? " " + pieza : " vacía")
          );

          if (pieza) {
            casilla.textContent = pieza;
            casilla.classList.add(
              blancas.includes(pieza) ? "blanca" : "negra"
            );
          }

          if (seleccion &&
              seleccion.fila === fila &&
              seleccion.col === col) {
            casilla.classList.add("seleccionada");
          }

          if (seleccion &&
              (seleccion.fila !== fila ||
               seleccion.col !== col)) {
            casilla.classList.add("posible");
            if (pieza) casilla.classList.add("ocupada");
          }

          if (col === 0) {
            const numero = document.createElement("span");
            numero.className = "coordenada fila";
            numero.textContent = 8 - fila;
            casilla.appendChild(numero);
          }

          if (fila === 7) {
            const letra = document.createElement("span");
            letra.className = "coordenada columna";
            letra.textContent = String.fromCharCode(97 + col);
            casilla.appendChild(letra);
          }

          casilla.addEventListener("click", () =>
            pulsar(fila, col)
          );

          tablero.appendChild(casilla);
        }
      }
    }

    function pulsar(fila, col) {
      const pieza = piezas[fila][col];

      if (!seleccion) {
        if (!pieza) {
          estado.textContent = "Selecciona una casilla con una pieza.";
          return;
        }

        seleccion = { fila, col };
        estado.textContent = "Ahora selecciona dónde moverla.";
      } else if (
        seleccion.fila === fila &&
        seleccion.col === col
      ) {
        seleccion = null;
        estado.textContent = "Selección cancelada.";
      } else {
        piezas[fila][col] =
          piezas[seleccion.fila][seleccion.col];
        piezas[seleccion.fila][seleccion.col] = "";

        seleccion = null;
        estado.textContent =
          "Pieza movida. Selecciona otra para continuar.";
      }

      dibujar();
    }

    document.getElementById("reiniciar").addEventListener(
      "click",
      () => {
        piezas = inicio.map(fila => [...fila]);
        seleccion = null;
        estado.textContent = "Tablero reiniciado.";
        dibujar();
      }
    );

    piezas = inicio.map(fila => [...fila]);
    dibujar();
  </script>
